📖 Python Quick Reference

Output & Input

print("Hello!")          # Display output
name = input("Name: ")   # Get user input

Variables & Math

x = 10                   # Integer
y = 3.14                 # Float
name = "Alice"           # String
is_valid = True          # Boolean

# Math operations
result = x + y           # Addition
result = x - y           # Subtraction
result = x * y           # Multiplication
result = x / y           # Division
result = x % y           # Modulus (remainder)

Conditionals

if x > 10:
    print("x is greater than 10")
elif x == 10:
    print("x equals 10")
else:
    print("x is less than 10")

Loops

# For loop
for i in range(5):
    print(i)

# While loop
count = 0
while count < 5:
    print(count)
    count += 1

Lists

numbers = [1, 2, 3, 4, 5]
numbers.append(6)        # Add to end
print(numbers[0])        # Access by index
print(len(numbers))      # Get length

for num in numbers:
    print(num)

Functions

def greet(name):
    return "Hello, " + name + "!"

result = greet("Alice")
print(result)

Math Module

import math

print(math.sqrt(16))     # Square root
print(math.pi)           # Pi constant
print(math.pow(2, 3))    # Power (2^3)

Random Module

import random

# Random integer
num = random.randint(1, 10)
print(num)

# Random choice from list
colors = ["red", "blue", "green"]
print(random.choice(colors))

# Random float between 0 and 1
print(random.random())
